cmake_minimum_required(VERSION 3.20)
project(drogon_live CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Drogon CONFIG REQUIRED)
add_library(feed src/feed.cc)
target_include_directories(feed PUBLIC src)
target_link_libraries(feed PUBLIC Drogon::Drogon)
target_compile_options(feed PRIVATE -Wall -Wextra -Werror)
add_executable(server src/main.cc)
target_link_libraries(server PRIVATE feed)
target_compile_options(server PRIVATE -Wall -Wextra -Werror)
include(CTest)
if(BUILD_TESTING)
  add_executable(feed_test test/feed_test.cc)
  target_link_libraries(feed_test PRIVATE feed)
  target_compile_options(feed_test PRIVATE -Wall -Wextra -Werror -UNDEBUG)
  add_test(NAME feed COMMAND feed_test)
endif()
